def get_current_ethiopian_context():
    # 1. Grab today's current Gregorian date and map it to Ethiopian Time
    today = Kenat.now()
    ethiopian_date = today.get_ethiopian() # e.g., {'year': 2018, 'month': 1, 'day': 17}
    
    # 2. Extract active regional or religious occasions for the current year
    current_holidays = get_holidays_for_year(ethiopian_date['year'])
